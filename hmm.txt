from django.db import models

#  Course Model
class Course(models.Model):
    name = models.CharField(max_length=100)
    duration = models.IntegerField()

    def __str__(self):
        return self.name


#  Skill Model (for multi checkbox)
class Skill(models.Model):
    name = models.CharField(max_length=100)

    def __str__(self):
        return self.name


#  Student Model
class Student(models.Model):

    # basic fields
    name = models.CharField(max_length=100)
    email = models.EmailField()
    date_of_birth = models.DateField()

    # file upload
    profile = models.FileField(upload_to='students/', null=True, blank=True)

    # dropdown
    COURSE_TYPE = [
        ('Regular', 'Regular'),
        ('Online', 'Online')
    ]
    course_type = models.CharField(max_length=20, choices=COURSE_TYPE, blank=False)

    # radio button
    GENDER = [
        ('Male', 'Male'),
        ('Female', 'Female')
    ]
    gender = models.CharField(max_length=10, choices=GENDER, blank=False)

    # checkbox
    is_active = models.BooleanField(default=True)

    # multi checkbox
    skills = models.ManyToManyField(Skill)

    # foreign key (dropdown)
    course = models.ForeignKey(Course, on_delete=models.CASCADE, blank=False)

    # number
    marks = models.IntegerField()

    def __str__(self):
        return self.name

from django.shortcuts import render, redirect
from .models import Student
from .forms import StudentForm


#  VIEW ALL STUDENTS (READ)
def view_student(request):
    data = Student.objects.all()
    return render(request, 'view.html', {'data': data})


# ADD STUDENT (CREATE)
def add_student(request):
    if request.method == "POST":
        form = StudentForm(request.POST, request.FILES)

        if form.is_valid():
            form.save()
            return redirect('view_student')
    else:
        form = StudentForm()

    return render(request, 'add.html', {'form': form})


#  EDIT STUDENT (UPDATE)
def edit_student(request, id):
    obj = Student.objects.get(id=id)

    if request.method == "POST":
        form = StudentForm(request.POST, request.FILES, instance=obj)

        if form.is_valid():
            form.save()
            return redirect('view_student')
    else:
        form = StudentForm(instance=obj)

    return render(request, 'add.html', {'form': form})


#  DELETE STUDENT
def delete_student(request, id):
    obj = Student.objects.get(id=id)
    obj.delete()
    return redirect('view_student')


from rest_framework.decorators import api_view
from rest_framework.response import Response
from .models import Student
from .serializers import StudentSerializer

@api_view(['GET'])
def get_students(request):
    data = Student.objects.all()
    serializer = StudentSerializer(data, many=True)
    return Response(serializer.data)

@api_view(['POST'])
def add_student_api(request):
    serializer = StudentSerializer(data=request.data)

    if serializer.is_valid():
        serializer.save()

    return Response(serializer.data)

from django.views.generic import ListView, CreateView, UpdateView, DeleteView
from django.urls import reverse_lazy
from .models import Student


#  LIST VIEW (READ)
class StudentListView(ListView):
    model = Student
    template_name = 'view.html'
    context_object_name = 'data'


#  CREATE VIEW (ADD)
class StudentCreateView(CreateView):
    model = Student
    fields = '__all__'
    template_name = 'add.html'
    success_url = reverse_lazy('view_student')


#  UPDATE VIEW (EDIT)
class StudentUpdateView(UpdateView):
    model = Student
    fields = '__all__'
    template_name = 'add.html'
    success_url = reverse_lazy('view_student')


#  DELETE VIEW
class StudentDeleteView(DeleteView):
    model = Student
    template_name = 'delete.html'
    success_url = reverse_lazy('view_student')
